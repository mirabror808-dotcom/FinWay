              <div>

                  { title: '50/30/20 Byudjetlashtirish Qoidasi', desc: 'Daromadning 50% qismini zaruriy xarajatlarga, 30% qismini xohish va ko\'ngilochar ehtiyojlarga, hamda 20% qismini jamg\'arma va investitsiyalarga yo\'naltirish tartibi.' },

                  </div>
                ))}
              </div>
            </div>

          </div>
        )}

      </main>

      {/* --- JUDGES PITCH MODAL --- */}
      {showPitchModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-950/80 backdrop-blur-md">

              <div className="p-3.5 rounded-xl bg-slate-950/70 border border-slate-800">
                <strong className="text-amber-400 font-bold block mb-1">1. Dolzarb Muammo:</strong>

            <button
              onClick={() => setShowPitchModal(false)}
